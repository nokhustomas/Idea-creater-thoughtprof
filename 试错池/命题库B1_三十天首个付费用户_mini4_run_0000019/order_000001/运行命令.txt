# 30秒自检命令（客户原话要求）
# 在工作目录下执行 bash 运行命令.txt 即可
# 功能：检查所有交付物存在 + grep 关键字段 + 验证 CSV 行数

set -e
cd "$(dirname "$0")"

# 1. 关键文件存在
ls README.md 用户调研报告.md 调研数据.csv 竞品定价表.md 竞品数据.csv \
   店家筛选标准.md 书面同意书.md 案例1_北京张姐快餐.md 案例2_广州老陈小吃.md \
   闲鱼商品文案.md 小红书笔记.md 备用引流方案.md \
   私域话术.md 朋友圈发布计划.md 群运营sop.md \
   报价单.md 服务合同.md 高意向用户筛选标准.md 退款申请表.md \
   复盘报告.md sop手册.md 数据记录表.md 运行命令.txt > /dev/null

# 2. 调研报告验收：含城市等级 + 20
grep -qE '(一线|新一线|二线|三线)' 用户调研报告.md
grep -q '20' 用户调研报告.md
[ "$(wc -l < 调研数据.csv)" -gt 20 ]

# 3. 竞品定价表验收：含全托/半托 + 来源平台 + 行数
grep -q '全托\|半托' 竞品定价表.md
grep -qE '(闲鱼|淘宝|抖音|美团)' 竞品定价表.md
[ "$(wc -l < 竞品数据.csv)" -gt 20 ]

# 4. 案例验收：≥2份 + 书面同意书 + 筛选标准 + 阅读量 + 评分
[ "$(ls 案例*.md | wc -l)" -ge 2 ]
ls 书面同意书.md 店家筛选标准.md > /dev/null
grep -q '阅读量' 案例*.md
grep -qE '3\.[5-9]|4\.' 店家筛选标准.md
grep -qE '[5-9][0-9]{2,}' 案例*.md

# 5. 引流验收：闲鱼 + 小红书 + 备用 + 微信路径
ls 闲鱼商品文案.md 小红书笔记.md 备用引流方案.md > /dev/null
grep -q '微信' 备用引流方案.md
grep -qE '代运营|引流|获客' 闲鱼商品文案.md 小红书笔记.md

# 6. 私域验收：3次/周 + 话术类型
ls 私域话术.md 朋友圈发布计划.md 群运营sop.md > /dev/null
grep -q '3次' 群运营sop.md
grep -qE '欢迎话术|跟进话术|转化话术' 私域话术.md

# 7. 付费转化验收：20% + 7天 + 文件齐全
ls 报价单.md 服务合同.md 高意向用户筛选标准.md 退款申请表.md > /dev/null
grep -qE '20%' 报价单.md 服务合同.md
grep -q '7天' 高意向用户筛选标准.md

# 8. 复盘验收：14天/20%/50%
ls 复盘报告.md sop手册.md 数据记录表.md > /dev/null
grep -qE '14天|20%|50%' 复盘报告.md

echo "✅ ALL CHECKS PASSED"
